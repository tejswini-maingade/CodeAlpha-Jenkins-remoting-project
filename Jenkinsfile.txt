pipeline {
    agent {
        label 'linux'
    }

    stages {

        stage('System Information') {
            steps {
                sh 'echo "Hostname:"'
                sh 'hostname'

                sh 'echo "Current User:"'
                sh 'whoami'

                sh 'echo "Operating System:"'
                sh 'uname -a'
            }
        }

        stage('Workspace Details') {
            steps {
                sh 'pwd'
                sh 'ls -la'
            }
        }

        stage('Verification') {
            steps {
                sh 'echo "🚀 Remote build executed successfully!"'
            }
        }
    }
}
